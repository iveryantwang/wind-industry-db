```dataview

  date AS "日期", link(file.link, title) AS "标题", region AS "区域",
  country AS "国别", segment AS "板块", importance AS "重要性", source_name AS "来源"
FROM "01-每日新闻"
WHERE type = "news"
SORT date DESC
LIMIT 200
```
